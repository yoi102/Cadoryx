cmake_minimum_required(VERSION 3.30)
project(CadoryxFiniteStandalonePrerequisite LANGUAGES CXX)
find_package(OpenCASCADE 8.0.1 EXACT REQUIRED PATHS "${OCCT_ROOT}/cmake" NO_DEFAULT_PATH)
add_executable(FiniteStandaloneProbe finite-standalone.cpp)
target_compile_features(FiniteStandaloneProbe PRIVATE cxx_std_20)
target_compile_definitions(FiniteStandaloneProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(FiniteStandaloneProbe PRIVATE /W4 /EHsc)
target_include_directories(FiniteStandaloneProbe PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
target_link_libraries(FiniteStandaloneProbe PRIVATE TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKShHealing TKPrim TKBO TKBool TKFeat)
